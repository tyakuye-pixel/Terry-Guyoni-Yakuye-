const SHEET_NAME = "Supporters";

function doGet(e) {
const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
const data = sheet.getDataRange().getValues();
const headers = data[0];
const rows = data.slice(1);

const supporters = rows.map((row, index) => {
let obj = { rowNumber: index + 2 }; // Track row position for edits/deletes
headers.forEach((header, i) => {
obj[header.toLowerCase()] = row[i];
});
return obj;
});

return ContentService.createTextOutput(JSON.stringify(supporters))
.setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
try {
const params = JSON.parse(e.postData.contents);
const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
const action = params.action;

if (action === "add") {
// Generate a unique ID using timestamp
const uniqueId = "SUP-" + new Date().getTime();
sheet.appendRow([params.name, params.province, params.llg, params.ward, params.role, uniqueId]);
return ContentService.createTextOutput(JSON.stringify({ success: true, id: uniqueId }))
.setMimeType(ContentService.MimeType.JSON);
}

if (action === "edit") {
const data = sheet.getDataRange().getValues();
for (let i = 1; i < data.length; i++) {
if (data[i][5] === params.id) { // Match Unique ID column
sheet.getRange(i + 1, 1, 1, 5).setValues([[params.name, params.province, params.llg, params.ward, params.role]]);
return ContentService.createTextOutput(JSON.stringify({ success: true }))
.setMimeType(ContentService.MimeType.JSON);
}
}
}

if (action === "delete") {
const data = sheet.getDataRange().getValues();
for (let i = 1; i < data.length; i++) {
if (data[i][5] === params.id) {
sheet.deleteRow(i + 1);
return ContentService.createTextOutput(JSON.stringify({ success: true }))
.setMimeType(ContentService.MimeType.JSON);
}
}
}

if (action === "clearAll") {
const lastRow = sheet.getLastRow();
if (lastRow > 1) {
sheet.getRange(2, 1, lastRow - 1, 6).clearContent();
}
return ContentService.createTextOutput(JSON.stringify({ success: true }))
.setMimeType(ContentService.MimeType.JSON);
}

} catch (error) {
return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString() }))
.setMimeType(ContentService.MimeType.JSON);
}
}

async function handleSupporterSubmission() {
const name = document.getElementById("supporterName").value.trim();
const province = document.getElementById("provinceSelect").value;
const llg = document.getElementById("llgSelect").value;
const ward = document.getElementById("wardNumber").value.trim();
const role = document.getElementById("roleInput").value.trim();

if (!name) { return alert("Please enter a name."); }
if (ward && (parseInt(ward) < 1 || parseInt(ward) > llgMaxWards[llg])) {
return alert(`Invalid Ward. ${llg} only has wards 1 to ${llgMaxWards[llg]}.`);
}

const wardFormatted = ward ? (ward.startsWith("Ward") ? ward : `Ward ${ward}`) : "N/A";
const payload = {
name: name,
province: province,
llg: llg,
ward: wardFormatted,
role: role || "Supporter"
};

// Duplicate Check
const isDuplicate = cachedSupporters.some(s => s.id !== editingSupporterId && s.name.toLowerCase() === name.toLowerCase() && s.llg === llg);
if (isDuplicate) {
if (!confirm(`⚠️ A supporter named "${name}" is already registered in ${llg}.\nDo you still want to proceed?`)) return;
}

document.getElementById("submitFormBtn").textContent = "⚡ Syncing...";
document.getElementById("submitFormBtn").disabled = true;

if (editingSupporterId) {
payload.action = "edit";
payload.id = editingSupporterId;
} else {
payload.action = "add";
}

try {
await fetch(API_URL, {
method: "POST",
body: JSON.stringify(payload)
});

// Reset inputs & state pointers
cancelFormEditing();
await fetchSupporters(); // Reload shared data matrix
} catch (error) {
alert("Error saving record to Google Sheet. Check connection.");
} finally {
document.getElementById("submitFormBtn").disabled = false;
}
}

function editSupporter(buttonElement) {

const targetId = buttonElement.getAttribute("data-id");
const currentData = cachedSupporters.find(s => s.id === targetId);

if (!currentData) return;

editingSupporterId = targetId;

document.getElementById("supporterName").value = currentData.name;
document.getElementById("provinceSelect").value = currentData.province || "Morobe";
document.getElementById("llgSelect").value = currentData.llg;
updateWardConstraints();

const numericWard = currentData.ward.replace("Ward ", "");
document.getElementById("wardNumber").value = numericWard === "N/A" ? "" : numericWard;
document.getElementById("roleInput").value = currentData.role === "Supporter" ? "" : currentData.role;

document.getElementById("submitFormBtn").textContent = "💾 Save Changes";
document.getElementById("cancelEditBtn").style.display = "inline-block";

}
